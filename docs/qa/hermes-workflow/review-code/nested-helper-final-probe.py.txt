"""Check real helper-group ownership when its launcher leader needs forced termination."""

from __future__ import annotations

import json
import signal
import sys
from pathlib import Path
from typing import Final

import pytest
from logopia_studio.launcher_process import run_process
from test_review_boundaries import process_state, settle_pid

REPO: Final = Path.cwd()


def test_forced_launcher_settles_nested_actual_helper(tmp_path: Path) -> None:
    # Given the actual helper wrapper inside a launcher leader that needs forced termination.
    leaf = tmp_path / "finite_leaf.py"
    leader = tmp_path / "finite_leader.py"
    pid_file = tmp_path / "helper.pid"
    _ = leaf.write_text(
        """import os, signal, sys
from pathlib import Path
signal.alarm(10)
Path(sys.argv[1]).write_text(str(os.getpid()))
signal.pause()
""",
        encoding="utf-8",
    )
    _ = leader.write_text(
        """import signal, sys
signal.alarm(10)
signal.signal(signal.SIGINT, signal.SIG_IGN)
sys.path.insert(0, sys.argv[3])
from logopia_studio.helper_process import run_cli
run_cli((sys.executable, sys.argv[1], sys.argv[2]), timeout=8)
""",
        encoding="utf-8",
    )
    # When the bounded real launcher exhausts grace and settles its leader's group.
    result = run_process(
        (sys.executable, str(leader), str(leaf), str(pid_file), str(REPO / "integrations/hermes")),
        tmp_path,
        tmp_path / "logs",
        2,
        grace_seconds=0.2,
    )
    pid = int(pid_file.read_text(encoding="utf-8"))
    try:
        observed = process_state(pid)
    finally:
        cleanup = settle_pid(pid)
    _ = sys.stdout.write(
        json.dumps(
            {
                "scenario": "forced_launcher_actual_helper",
                "launcher_exit": result.exit_code,
                "timed_out": result.timed_out,
                "helper_pid": pid,
                "helper_after_receipt": observed,
                "cleanup": cleanup,
            }
        )
        + "\n"
    )
    # Then that invocation's helper must also be settled before a receipt is returned.
    assert result.timed_out
    assert not observed, f"Actual helper survived its forced launcher: {observed}"
    assert cleanup == "absent"


if __name__ == "__main__":
    _ = signal.alarm(20)
    raise SystemExit(
        pytest.main(
            [
                "-q",
                "-s",
                "-c",
                "pyproject.toml",
                __file__,
                "--basetemp",
                str(Path(__file__).with_name("nested-pytest")),
            ]
        )
    )
